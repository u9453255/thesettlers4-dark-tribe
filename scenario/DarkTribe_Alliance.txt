-- The Alliance v0.2.15. Native farm workers handle mushrooms; the plugin handles paid construction.
-- Romans, Vikings and Mayas are allied computer players authored by the source map; no timed resources or waves.
DT_Version = 200
DT_Scenario = 3
DT_IsNew = 0
DT_VictoryArmed = 0
DT_StartWarning = 0
-- The game's old Lua has no numeric loops: one explicit line per opponent.
function DT_Towers(p)
  return Buildings.Amount(p, Buildings.GUARDTOWERSMALL, Buildings.READY)
         + Buildings.Amount(p, Buildings.GUARDTOWERBIG, Buildings.READY)
         + Buildings.Amount(p, Buildings.CASTLE, Buildings.READY)
end
function DT_EnemyBases()
  return DT_Towers(2) + DT_Towers(3) + DT_Towers(4)
end
function DT_Aggressive(p)
  AI.SetPlayerVar(p, "AttackMode", 2, 3, 3)
  AI.SetPlayerVar(p, "AttackDelay", 40, 20, 10)
  AI.SetPlayerVar(p, "SoldierLimitAbsolute", 100, 200, 300)
  AI.SetPlayerVar(p, "SoldierLimitRelative", 100, 150, 200)
end
function DT_Init()
  DT_IsNew = 1
  -- No Mushroom Farm stands yet: enough mana for the first farms.
  Magic.IncreaseMana(1, 150)
  Map.SetScreenPos(332, 216)
  -- Original AI of each people owns construction, production, recruitment and attacks.
  -- Values are within the range of the shipped single-player scripts, at their aggressive end.
  DT_Aggressive(2)
  DT_Aggressive(3)
  DT_Aggressive(4)
  dbg.stm("THE ALLIANCE: Romans, Vikings and Mayas have joined forces against you. Destroy every tower of all three peoples.")
  dbg.stm("Your Temple stands alone and you have no farms yet: build Mushroom Farms on Dark land first.")
  dbg.stm("If every Temple and Fortress falls, the Dark Tribe is lost.")
end
function DT_Resume()
  -- Nothing to announce when a game is loaded.
end
function DT_Victory()
  local darkBases = Buildings.Amount(1, Buildings.DARKTEMPLE, Buildings.READY)
                    + Buildings.Amount(1, 51, Buildings.READY)
  local enemyBases = DT_EnemyBases()
  -- A zero count before map initialization is not a defeated player.
  -- Arm once both genuine bases have been seen; afterwards destruction is decisive.
  if DT_VictoryArmed == 0 then
    if darkBases > 0 and enemyBases > 0 then
      DT_VictoryArmed = 1
    else
      -- A base can be missing only before map initialization; never announced to the player.
      DT_StartWarning = 1
      return
    end
  end
  if darkBases == 0 then
    Game.EnemyPlayersLost(2)
  elseif enemyBases == 0 then
    Game.EnemyPlayersLost(1)
  else
    -- Native default end logic also evaluates inherited mission objectives.
    -- Consult it only after this scenario has explicitly declared a loser.
    return
  end
  Game.DefaultGameEndCheck()
end
function new_game()
  request_event(DT_Init, Events.FIRST_TICK_OF_NEW_GAME)
  request_event(DT_Resume, Events.FIRST_TICK_OF_NEW_OR_LOADED_GAME)
  request_event(DT_Victory, Events.VICTORY_CONDITION_CHECK)
end
function register_functions()
  reg_func(DT_Init)
  reg_func(DT_Resume)
  reg_func(DT_Victory)
end
